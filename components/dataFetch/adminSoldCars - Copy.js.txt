import React, { useState, useEffect, useRef } from 'react';
import { useUser } from '../user/userContext';

const AdminSoldCars = () => {
  const { user } = useUser();
  const [cars, setCars] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [selectedImage, setSelectedImage] = useState(null);
  const [expandedCar, setExpandedCar] = useState(null);
  const [editingCar, setEditingCar] = useState(null);
  const [editFormData, setEditFormData] = useState({});
  const [successMessage, setSuccessMessage] = useState('');
  const fileInputRef = useRef(null);

  // apiUrl is for API endpoints (e.g., PHP scripts)
  const apiUrl = process.env.NODE_ENV === 'development'
    ? 'http://localhost/artisbay-next/server'
    : '/server';
  // imgBasePath is the base URL for publicly accessible images
  const imgBasePath = process.env.NODE_ENV === 'development'
    ? 'http://localhost/artisbay-next/server'
    : '/server';

  useEffect(() => {
    const fetchData = async () => {
      try {
        const response = await fetch(`${apiUrl}/fetchAllSoldCars.php`, {
          credentials: 'include'
        });
        
        if (!response.ok) {
          const errorText = await response.text();
          throw new Error(errorText || 'Failed to fetch cars');
        }
        
        const result = await response.json();
        if (!result.success) throw new Error(result.error || 'Invalid data');
        
        // Assume result.data.image_urls already contain relative paths like '/uploads/car_folder/image.png'
        setCars(result.data.map(car => ({
          ...car,
          image_urls: Array.isArray(car.image_urls) ? car.image_urls : []
        })) || []);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, [apiUrl]);

  // Clean up object URLs when component unmounts
  useEffect(() => {
    return () => {
      if (editFormData.image_urls) {
        editFormData.image_urls.forEach(img => {
          if (img.preview) URL.revokeObjectURL(img.preview);
        });
      }
    };
  }, [editFormData]);

  const toggleExpand = (carId) => {
    setExpandedCar(expandedCar === carId ? null : carId);
  };

  const handleEditClick = (car) => {
    setEditingCar(car.id);
    setEditFormData({
      ...car,
      // image_urls from car object should already be relative paths for display.
      image_urls: Array.isArray(car.image_urls) ? [...car.image_urls] : []
    });
  };

  const handleCancelEdit = () => {
    setEditingCar(null);
    setSelectedImage(null);
    setSuccessMessage('');
  };

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    setEditFormData(prev => ({
      ...prev,
      [name]: value
    }));
  };

  const handleImageUpload = (e) => {
    const files = Array.from(e.target.files);
    if (files.length === 0) return;

    const newImages = files.map(file => ({
      name: file.name,
      file, // Store file object for upload
      preview: URL.createObjectURL(file) // Create preview URL for immediate display
    }));

    setEditFormData(prev => ({
      ...prev,
      image_urls: [...prev.image_urls, ...newImages]
    }));
  };

  const handleRemoveImage = (index) => {
    setEditFormData(prev => {
      const updatedImages = [...prev.image_urls];
      // Revoke object URL if it's a preview (newly added image)
      if (updatedImages[index]?.preview) {
        URL.revokeObjectURL(updatedImages[index].preview);
      }
      updatedImages.splice(index, 1);
      return {
        ...prev,
        image_urls: updatedImages
      };
    });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError(null); // Clear previous errors
    try {
      const formData = new FormData();
      
      // Append all non-image fields
      Object.entries(editFormData).forEach(([key, value]) => {
        if (key !== 'image_urls' && key !== 'preview' && value !== undefined) {
          formData.append(key, value);
        }
      });

      // Handle existing images - send paths *as they are* (e.g., '/uploads/car_folder/image.png')
      const existingImagesToSend = editFormData.image_urls
        .filter(img => typeof img === 'string'); // Filter out new image objects

      formData.append('existing_images', JSON.stringify(existingImagesToSend));

      // Send original images for comparison *as they are*
      const originalCar = cars.find(c => c.id === editingCar);
      const originalImagesToSend = (originalCar.image_urls || []);

      formData.append('original_images', JSON.stringify(originalImagesToSend));

      // Handle new image uploads
      editFormData.image_urls
        .filter(img => img.file) // Filter for new image objects that have a 'file' property
        .forEach((img, index) => {
          formData.append(`new_images[${index}]`, img.file, img.name);
        });

      const response = await fetch(`${apiUrl}/updateSoldCar.php`, {
        method: 'POST',
        credentials: 'include',
        body: formData
      });

      if (!response.ok) {
        const errorText = await response.text();
        throw new Error(errorText || 'Update failed');
      }

      const data = await response.json();
      if (data.error) throw new Error(data.error);

      // Update state with the new list of image URLs from the server.
      // The server should return paths like '/uploads/car_folder/image.png'
      setCars(cars.map(car => 
        car.id === editingCar ? { 
          ...editFormData,
          image_urls: data.image_urls // These are now '/uploads/car_folder/image.png'
        } : car
      ));
      
      setEditingCar(null);
      setSuccessMessage('Car updated successfully!');
      setTimeout(() => setSuccessMessage(''), 3000);
    } catch (err) {
      setError(err.message);
      console.error('Update error:', err);
    }
  };
  
  if (loading) return <div className="p-4">Loading...</div>;
  if (error) return <div className="p-4 text-red-500">Error: {error}</div>;

  return (
    <div className="p-4">
      <h2 className="text-2xl font-bold mb-6">Sold Cars Management</h2>
      
      {successMessage && (
        <div className="mb-4 p-3 bg-green-100 text-green-700 rounded">
          {successMessage}
        </div>
      )}

      <div className="overflow-x-auto">
        <table className="min-w-full bg-white rounded-lg">
          <thead className="bg-gray-100">
            <tr>
              <th className="p-3 text-left">Ref No</th>
              <th className="p-3 text-left">Make/Model</th>
              <th className="p-3 text-left">Year</th>
              <th className="p-3 text-left">Price</th>
              <th className="p-3 text-left">Images</th>
              <th className="p-3 text-left">Actions</th>
            </tr>
          </thead>
          <tbody>
            {cars.map((car) => (
              <React.Fragment key={car.id}>
                {/* Regular row - show this when NOT editing */}
                {editingCar !== car.id && (
                  <>
                    <tr className="border-b hover:bg-gray-50">
                      <td className="p-3">
                        {car.ref_no}
                      </td>
                      <td className="p-3">
                        {car.make} {car.model}
                      </td>
                      <td className="p-3">
                        {car.year}
                      </td>
                      <td className="p-3">
                        {car.price} {car.currency}
                      </td>
                      <td className="p-3">
                        <div className="flex space-x-2">
                          {car.image_urls?.slice(0, 3).map((img, index) => {
                            // img from car.image_urls is like '/uploads/car_folder/image.png'
                            return (
                              <img 
                                key={index}
                                src={
                                  typeof img === 'string' 
                                    ? `${imgBasePath}/${img.startsWith('/') ? img.substring(1) : img}`
                                    : img.preview // For newly added images before save
                                } 
                                className="w-12 h-12 object-cover cursor-pointer"
                                onClick={() => setSelectedImage(img)} // Store relative path for modal
                                alt={`Thumbnail ${index}`}
                              />
                            );
                          })}
                          {car.image_urls?.length > 3 && (
                            <div className="w-12 h-12 bg-gray-200 flex items-center justify-center">
                              +{car.image_urls.length - 3}
                            </div>
                          )}
                        </div>
                      </td>
                      <td className="p-3 space-x-2">
                        <button
                          onClick={() => toggleExpand(car.id)}
                          className="bg-gray-500 text-white px-3 py-1 rounded hover:bg-gray-600"
                        >
                          {expandedCar === car.id ? 'Hide' : 'Details'}
                        </button>
                        <button
                          onClick={() => handleEditClick(car)}
                          className="bg-blue-500 text-white px-3 py-1 rounded hover:bg-blue-600"
                        >
                          Edit
                        </button>
                      </td>
                    </tr>
                    {expandedCar === car.id && (
                      <tr className="border-b bg-gray-50">
                        <td colSpan="6" className="p-4">
                          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                            <div>
                              <h4 className="font-bold">Basic Info</h4>
                              <p>Size: {car.size}</p>
                              <p>Category: {car.category}</p>
                              <p>Color: {car.color}</p>
                              <p>Dimension: {car.dimension}</p>
                              <p>M3: {car.m3}</p>
                            </div>
                            <div>
                              <h4 className="font-bold">Technical</h4>
                              <p>Engine: {car.engine_capacity}</p>
                              <p>Mileage: {car.mileage}</p>
                              <p>Chassis: {car.chassis_no}</p>
                              <p>Fuel: {car.fuel}</p>
                              <p>Doors: {car.door}</p>
                              <p>Seats: {car.seat}</p>
                            </div>
                            <div>
                              <h4 className="font-bold">Shipping</h4>
                              <p>Ship: {car.ship_name}</p>
                              <p>Departure: {car.departure_port}</p>
                              <p>Arrival: {car.arrival_port}</p>
                              <p>Freight: {car.freight}</p>
                              <p>Discount: {car.discount}</p>
                              <p>Final Value: {car.final_value}</p>
                            </div>
                          </div>
                        </td>
                      </tr>
                    )}
                  </>
                )}
                
                {/* Edit form row - shown when editing */}
                {editingCar === car.id && (
                  <tr className="bg-blue-50">
                    <td colSpan="6" className="p-4">
                      <form onSubmit={handleSubmit} className="space-y-4">
                        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                          {/* Basic Info */}
                          <div className="space-y-2">
                            <h4 className="font-bold">Basic Info</h4>
                            <div>
                              <label className="block text-sm font-medium">Ref No</label>
                              <input type="text" name="ref_no" value={editFormData.ref_no || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Make</label>
                              <input type="text" name="make" value={editFormData.make || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Model</label>
                              <input type="text" name="model" value={editFormData.model || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Year</label>
                              <input type="text" name="year" value={editFormData.year || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Price</label>
                              <input type="text" name="price" value={editFormData.price || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Currency</label>
                              <input type="text" name="currency" value={editFormData.currency || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                          </div>

                          {/* Technical Details */}
                          <div className="space-y-2">
                            <h4 className="font-bold">Technical</h4>
                            <div>
                              <label className="block text-sm font-medium">Size</label>
                              <input type="text" name="size" value={editFormData.size || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Category</label>
                              <input type="text" name="category" value={editFormData.category || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Color</label>
                              <input type="text" name="color" value={editFormData.color || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Dimension</label>
                              <input type="text" name="dimension" value={editFormData.dimension || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">M3</label>
                              <input type="text" name="m3" value={editFormData.m3 || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Engine Capacity</label>
                              <input type="text" name="engine_capacity" value={editFormData.engine_capacity || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                          </div>

                          {/* Shipping Details */}
                          <div className="space-y-2">
                            <h4 className="font-bold">Shipping</h4>
                            <div>
                              <label className="block text-sm font-medium">Ship Name</label>
                              <input type="text" name="ship_name" value={editFormData.ship_name || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Departure Port</label>
                              <input type="text" name="departure_port" value={editFormData.departure_port || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Arrival Port</label>
                              <input type="text" name="arrival_port" value={editFormData.arrival_port || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Freight</label>
                              <input type="text" name="freight" value={editFormData.freight || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Discount</label>
                              <input type="text" name="discount" value={editFormData.discount || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                            <div>
                              <label className="block text-sm font-medium">Final Value</label>
                              <input type="text" name="final_value" value={editFormData.final_value || ''} onChange={handleInputChange} className="w-full border rounded p-2" />
                            </div>
                          </div>
                        </div>

                        {/* Image Management Section */}
                        <div className="mb-4">
                          <h3 className="font-bold mb-2">Car Images</h3>
                          <div className="flex flex-wrap gap-4 mb-4">
                            {editFormData.image_urls?.map((img, index) => (
                              <div key={index} className="relative group">
                                <img
                                  // For existing images, img will be a string (e.g., '/uploads/car_folder/image.png').
                                  // For newly uploaded images (before save), img will be an object with 'preview'.
                                  src={
                                    typeof img === 'string' 
                                      ? `${imgBasePath}/${img.startsWith('/') ? img.substring(1) : img}`
                                      : img.preview
                                  }
                                  className="w-32 h-32 object-cover rounded border"
                                  alt={`Car image ${index}`}
                                />
                                <button
                                  type="button"
                                  onClick={() => handleRemoveImage(index)}
                                  className="absolute top-0 right-0 bg-red-500 text-white rounded-full w-6 h-6 flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity"
                                >
                                  ×
                                </button>
                              </div>
                            ))}
                          </div>
                          
                          <div>
                            <button
                              type="button"
                              onClick={() => fileInputRef.current.click()}
                              className="bg-gray-200 hover:bg-gray-300 px-4 py-2 rounded"
                            >
                              Add Images
                            </button>
                            <input
                              type="file"
                              ref={fileInputRef}
                              onChange={handleImageUpload}
                              multiple
                              accept="image/*"
                              className="hidden"
                            />
                          </div>
                        </div>

                        <div className="flex justify-end space-x-4">
                          <button
                            type="button"
                            onClick={handleCancelEdit}
                            className="bg-gray-500 text-white px-4 py-2 rounded hover:bg-gray-600"
                          >
                            Cancel
                          </button>
                          <button
                            type="submit"
                            className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
                          >
                            Save Changes
                          </button>
                        </div>
                      </form>
                    </td>
                  </tr>
                )}
              </React.Fragment>
            ))}
          </tbody>
        </table>
      </div>

      {/* Image modal */}
      {selectedImage && (
        <div className="fixed inset-0 bg-black/75 flex items-center justify-center z-50 p-4">
          <div className="bg-white rounded-lg max-w-4xl max-h-[90vh] overflow-auto">
            <div className="flex justify-between items-center p-4 border-b">
              <h3 className="text-lg font-semibold">Car Image</h3>
              <button
                onClick={() => setSelectedImage(null)}
                className="text-2xl hover:text-gray-700"
              >
                &times;
              </button>
            </div>
            <div className="p-4">
              <img
                src={
                  typeof selectedImage === 'string' 
                    ? `${imgBasePath}/${selectedImage.startsWith('/') ? selectedImage.substring(1) : selectedImage}` 
                    : selectedImage.preview
                } 
                className="max-w-full max-h-[80vh] mx-auto"
                alt="Full size car"
              />
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default AdminSoldCars;
